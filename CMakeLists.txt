cmake_minimum_required(VERSION 3.20)
project(CacheForge VERSION 0.1.0 LANGUAGES CXX)

add_library(cfdb STATIC src/engine.cpp src/wal.cpp src/checksum.cpp)
target_include_directories(cfdb PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(cfdb PUBLIC cxx_std_20)
set_target_properties(cfdb PROPERTIES CXX_EXTENSIONS OFF)

if(MSVC)
    target_compile_options(cfdb PRIVATE /W4)
else()
    target_compile_options(cfdb PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(engine_test tests/engine_test.cpp)
    target_link_libraries(engine_test PRIVATE cfdb)
    set_target_properties(engine_test PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(engine_test PRIVATE /W4)
    else()
        target_compile_options(engine_test PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME engine_contract COMMAND engine_test)

    add_executable(wal_test tests/wal_test.cpp)
    target_link_libraries(wal_test PRIVATE cfdb)
    set_target_properties(wal_test PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(wal_test PRIVATE /W4)
    else()
        target_compile_options(wal_test PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME wal_contract COMMAND wal_test)

    add_executable(recovery_test tests/recovery_test.cpp)
    target_link_libraries(recovery_test PRIVATE cfdb)
    set_target_properties(recovery_test PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(recovery_test PRIVATE /W4)
    else()
        target_compile_options(recovery_test PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME recovery_contract COMMAND recovery_test)

    add_executable(checksum_test tests/checksum_test.cpp)
    target_link_libraries(checksum_test PRIVATE cfdb)
    set_target_properties(checksum_test PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(checksum_test PRIVATE /W4)
    else()
        target_compile_options(checksum_test PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME checksum_contract COMMAND checksum_test)
endif()
